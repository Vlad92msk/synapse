import { MemoryStorage } from 'synapse-storage/core'

const todoStorage = new MemoryStorage<TodoState>({

name: 'todo',

initialState: { todos: [], filter: 'all' },

})

await todoStorage.initialize()

todoStorage.set('filter', 'active')

todoStorage.update((s) => {          // «мутируем», как в Immer

s.todos.push({ id: 't3', title: 'Снять видео', done: false })

s.filter = 'all'

})                                   // → одно уведомление

Начнём с самого нижнего кирпича — хранилища. Импортируем MemoryStorage из synapse-storage/core. Создаём экземпляр. Нужны имя и начальное состояние. Тип состояния выводится из initialState или задаётся дженериком. Перед работой хранилище инициализируем. У всех хранилищ это одинаковый шаг: персистентные в этот момент подтягивают сохранённые данные. Писать можно двумя способами. set меняет одно поле. update позволяет „мутировать“ черновик прямо в коллбэке, как в Immer. Неизменяемое состояние библиотека соберёт сама, а подписчики получат одно уведомление на весь коллбэк. Подписаться можно на ключ, на вычисляемое значение — тогда коллбэк сработает, только когда это значение изменилось, или на любые изменения сразу. В событии придут пути полей, которые поменялись. Это удобно для логирования и синхронизации. Хранилище работает и на сервере, и без React.
